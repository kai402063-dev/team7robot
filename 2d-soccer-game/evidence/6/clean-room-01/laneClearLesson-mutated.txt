function clearLane = laneClearLesson(a, b, obstacles, margin)
% Independent exercise from DEV_MANUAL -> chapters/handson.md path 1.
% Preserve the linked tdp.laneClear policy for a degenerate segment.
segment = b - a;
lengthSquared = dot(segment, segment);
clearLane = true;
if lengthSquared < 1e-9
    return
end
for i = 1:size(obstacles, 1)
    fraction = dot(obstacles(i,:) - a, segment) / lengthSquared;
    fraction = min(1, max(0, fraction));
    nearest = a + fraction * segment;
    distance = norm(obstacles(i,:) - nearest);
    % Equality is permitted: the linked implementation uses strict <.
    if distance > margin
        clearLane = false;
        return
    end
end
end
